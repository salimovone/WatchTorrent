cmake_minimum_required(VERSION 3.24)
project(watchtorrent_libtorrent LANGUAGES CXX)
find_package(LibtorrentRasterbar CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_library(watchtorrent_libtorrent STATIC bridge.cpp)
else()
  add_library(watchtorrent_libtorrent SHARED bridge.cpp)
endif()
target_compile_features(watchtorrent_libtorrent PRIVATE cxx_std_17)
target_link_libraries(watchtorrent_libtorrent PRIVATE LibtorrentRasterbar::torrent-rasterbar nlohmann_json::nlohmann_json)
set_target_properties(watchtorrent_libtorrent PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
  target_link_options(watchtorrent_libtorrent PRIVATE "-Wl,-z,max-page-size=16384")
endif()
install(TARGETS watchtorrent_libtorrent RUNTIME DESTINATION . LIBRARY DESTINATION . ARCHIVE DESTINATION .)
include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_executable(libtorrent_loopback tests.cpp)
  target_compile_features(libtorrent_loopback PRIVATE cxx_std_17)
  target_link_libraries(libtorrent_loopback PRIVATE LibtorrentRasterbar::torrent-rasterbar nlohmann_json::nlohmann_json)
  add_test(NAME verified_loopback_download COMMAND libtorrent_loopback)
  set_tests_properties(verified_loopback_download PROPERTIES TIMEOUT 60)
endif()
